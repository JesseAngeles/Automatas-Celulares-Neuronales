\section{Objetivos}

\begin{frame}{Objetivo}
	\begin{center}
		\begin{tikzpicture}
			\node[cajaobjetivo, text width=0.85\textwidth]
			{
				Desarrollar un sistema de simulación basado en autómatas celulares neuronales
				bidimensionales que permita analizar la generación, mantenimiento, recuperación
				y propagación de patrones de información en entornos complejos y dinámicos.
			};
		\end{tikzpicture}
	\end{center}
\end{frame}

\begin{frame}{Objetivos específicos}
	\begin{center}
		\begin{tikzpicture}
			\node[cajaobjetivo, text width=0.9\textwidth]
			{

			};
		\end{tikzpicture}
	\end{center}
	
	\begin{center}
		\begin{tikzpicture}

			};
		\end{tikzpicture}
	\end{center}
	
	\begin{center}
		\begin{tikzpicture}
			\node[cajaobjetivo, text width=0.9\textwidth]

		\end{tikzpicture}
	\end{center}
